<#
.SYNOPSIS
    This script start the data collection interface for the Microsoft Network Performance Assessment.

.DESCRIPTION
    The Start-Datacollection.ps1 script is used to start the data collection interface for the Microsoft Network Performance Assessment.
    It will check the necessary prerequisites before starting the interface.

.PARAMETER TenantType
    Specifies the type of tenant. Valid values are "Worldwide", "GCCH", and "DOD".
    The default value is "Worldwide".
    Example: -TenantType "GCCH" > This will set the script for a GCCH tenant.

.PARAMETER TenantName
    Specifies the name of the tenant to use for SPO connections.
    By default the script will use "microsoft-my.sharepoint.com".
    Example: -TenantName "contoso" > This will connect to the "contoso-my.sharepoint.com" SPO FQDN.

.PARAMETER SetProxyConfiguration
    Switch to start a wizard to reconfigure proxy settings on the current user.
    Example: -SetProxyConfiguration > This will start the proxy configuration wizard.

.PARAMETER SetProxyCredentials
    Switch to explicitly enter the user credentials to use for proxy authentication (if proxy authentication is not explicitly detected).
    Example: -SetProxyCredentials > This will prompt to enter the user credentials for proxy authentication.

.EXAMPLE
    .\Start-Datacollection.ps1 -TenantName "contoso"

    This command will run the script and attempt to connect to the "contoso-my.sharepoint.com" SPO FQDN.

.EXAMPLE
    .\Start-Datacollection.ps1 -SetProxyConfiguration

    This command will run the script with the default configuration and start the proxy configuration wizard.

.EXAMPLE
    .\Start-Datacollection.ps1 -TenantType "GCCH" -TenantName "contoso" -SetProxyConfiguration

    This command will run the script for a GCCH tenant, start the proxy configuration wizard, and attempt to connect to the "contoso-my.sharepoint.us" SPO FQDN.

#>

    }

			} else {

                $CurrentSettings.OneDriveFQDN = $Endpoints.Sharepoint
                Write-Log -Message "Sharepoint FQDN built using tenant name '$($Endpoints.Sharepoint)'" -Source "Init" -MessageType Info
            }
			
        }

        Write-Log -Message "Fetching endpoints for GCCH tenants" -Source "Init" -MessageType Info

            }

### $ScriptsPath = "xxxx" ; Import-Module "$ScriptsPath\Scripts\DataCollection.psm1" ; Get-ChildItem "$ScriptsPath\Start*","$ScriptsPath\Scripts\*" -Exclude "FileHashes.json","*.exe" | Get-FileContentHash | Select-Object -Property @{Name="Filename";Expression={Split-Path $_.Path -Leaf}},Hash | ConvertTo-Json | Out-File -FilePath "$ScriptsPath\Scripts\FileHashes.json" -Encoding UTF8 -Force

$ExpectedFileHashes = Get-Content "$env:NPAPath\Scripts\FileHashes.json" | ConvertFrom-Json

$FileIntegrityIssues = @()

    $CorrespondingHash = $CurrentFileHashes | Where-Object {$_.Filename -eq $ExpectedFileHash.Filename}
    if ($CorrespondingHash) {
        if ($CorrespondingHash.Hash -ne $ExpectedFileHash.Hash) {
            Write-Log -Message "File '$($ExpectedFileHash.Filename)' has been altered" -Source "Init" -MessageType Warning
            $FileIntegrityIssues += "Warning"
        }
    } else {
        Write-Log -Message "File '$($ExpectedFileHash.Filename)' is missing" -Source "Init" -MessageType Error
        $FileIntegrityIssues += "Error"
    }
}
if ($FileIntegrityIssues) {
    if ($FileIntegrityIssues -match "Error") {
        Write-Host "FAILED" -ForegroundColor Red
        Write-Warning "Found at least one major file integrity issue`nThis will cause issues"
        Write-Log -Message "Found at least one major file integrity issue. Exiting..." -Source "Init" -MessageType Error

        exit

    } else {
        Write-Host "WARNING" -ForegroundColor Yellow
        Write-Warning "Found at least one file integrity issue"
        Write-Log -Message "Found at least one file integrity issue" -Source "Init" -MessageType Warning

Write-Host "Checking DataCollection version - " -NoNewline
Write-Log -Message "Checking expected DataCollection version information" -Source "Init" -MessageType Info
try {
    $NPAVersionCheck = ((Invoke-WebRequest -Uri 'https://forms.office.com/handlers/ResponsePageStartup.ashx?id=v4j5cvGGr0GRqy180BHbR6wEhZooLZxDgzjnVWKYS0JUMEFTQ0w4QUhNRDRXVzhFOUgzTjFIMFFGRCQlQCN0PWcu' -UserAgent (Get-UserAgent) -UseBasicParsing).Content | ConvertFrom-Json)
    $ExpectedVersion = (($NPAVersionCheck.data.form.questions | Where-Object {$_.title -eq "NPA versions"}).Choices.DisplayText | Where-Object {$_ -match "DataCollection - "}) -replace '.*? - (.*)','$1'
    if ($ExpectedVersion -notmatch "\d.\d") { throw "No valid version string fetched" }
    if ([System.Version]($Version -replace '(\d.\d\.\d).*','$1') -lt [System.Version]$ExpectedVersion) {
        Write-Log -Message "Using an old version of DataCollection (Current: '$Version' - Expected: '$ExpectedVersion')" -Source "Init" -MessageType Error
        if ($BypassVersionCheck) {
            Write-Log -Message "Bypassing version check" -Source "Init" -MessageType Warning
            Write-Host "WARNING" -ForegroundColor Yellow
        } else {
            Write-Host "FAILED" -ForegroundColor Red
            $Breakflag = $true
        }
        Write-Warning "A newer version of the DataCollection tool is available (Current: '$Version' - Expected: '$ExpectedVersion')"
        $VersionMatch = $false
    } elseif ([System.Version]($Version -replace '(\d.\d\.\d).*','$1') -eq [System.Version]$ExpectedVersion) {
        Write-Log -Message "Using the expected version of DataCollection ('$ExpectedVersion')" -Source "Init" -MessageType Info
        Write-Host "PASSED" -ForegroundColor Green
        $VersionMatch = $true
    } else {
        Write-Log -Message "Using a different version of DataCollection (Current: '$Version' - Expected: '$ExpectedVersion')" -Source "Init" -MessageType Warning
        Write-Host "PASSED" -ForegroundColor Green
        $VersionMatch = $false
    }
} catch {
    Write-Log -Message "Failed to check expected DataCollection version information: $($_.Exception.Message)" -Source "Init" -MessageType Error
    Write-Host "WARNING" -ForegroundColor Yellow
    Write-Warning "Unable to check for the expected version of the DataCollection tool"
}

# Only check for known issues if the version is the expected one
# This is to avoid false "known issues" when the versions don't match
if ($VersionMatch -eq $true) {
    Write-Host "Checking DataCollection known issues - " -NoNewline
    Write-Log -Message "Checking for known issues" -Source "Init" -MessageType Info
    try {
        $KnownIssuesCheck = ((Invoke-WebRequest -Uri 'https://forms.office.com/handlers/ResponsePageStartup.ashx?id=v4j5cvGGr0GRqy180BHbR6wEhZooLZxDgzjnVWKYS0JUMEFTQ0w4QUhNRDRXVzhFOUgzTjFIMFFGRCQlQCN0PWcu' -UserAgent (Get-UserAgent) -UseBasicParsing).Content | ConvertFrom-Json)
        $KnownIssues = (($KnownIssuesCheck.data.form.questions | Where-Object {$_.title -eq "Known issues"}).Choices.DisplayText | Where-Object {$_ -match "DataCollection - "}) -replace '.*? - (.*)','$1'
        if ($KnownIssues -eq "None") {
            Write-Log -Message "No known issues found" -Source "Init" -MessageType Info
            Write-Host "PASSED" -ForegroundColor Green
        } else {
            Write-Log -Message "Known issues found" -Source "Init" -MessageType Warning
            Write-Host "WARNING" -ForegroundColor Yellow
            foreach ($KnownIssue in ($KnownIssues -split ';')) {
                Write-Log -Message "Issue: $KnownIssue" -Source "Init" -MessageType Warning
                Write-Warning "Issue: $KnownIssue"
            }
        }
    } catch {
        Write-Log -Message "Unable to check for known issues: $($_.Exception.Message)" -Source "Init" -MessageType Error
        Write-Host "WARNING" -ForegroundColor Yellow
        Write-Warning "Unable to check for known issues of the DataCollection tool"
    }
} else {
    Write-Log -Message "Not checking for known issues due to version mismatch" -Source "Init" -MessageType Warning
}

Write-Host "Checking installed capture tools - " -NoNewline
Write-Log -Message "Checking installed capture tools" -Source "Init" -MessageType Info

Write-Log -Message "Checking Wireshark" -Source "Init" -MessageType Info
if (Test-Path HKLM:\Software\Wow6432Node\Microsoft\Windows\CurrentVersion\Uninstall\Wireshark) {
    $WiresharkPath = (Get-ItemProperty HKLM:\Software\Wow6432Node\Microsoft\Windows\CurrentVersion\Uninstall\Wireshark -Name InstallLocation).InstallLocation
} elseif (Test-Path HKLM:\Software\Microsoft\Windows\CurrentVersion\Uninstall\Wireshark) {
    $WiresharkPath = (Get-ItemProperty HKLM:\Software\Microsoft\Windows\CurrentVersion\Uninstall\Wireshark -Name InstallLocation).InstallLocation
} elseif (Test-Path "C:\program files\Wireshark\tshark.exe") {
    $WiresharkPath = "C:\program files\Wireshark"
} elseif (Test-Path "C:\Program Files (x86)\Wireshark\tshark.exe") {
    $WiresharkPath = "C:\Program Files (x86)\Wireshark"
} else {
    $WiresharkPath = $null
}

    if ($KnownIssues -notlike "" -and $KnownIssues -ne "None") {
        Write-Log -Message "Displaying popup warning for known issues" -Source "Init" -MessageType Warning
        $WarningTitle = "Known issues found"
        $WarningText = "Current known issues:`r`n"
        foreach ($KnownIssue in ($KnownIssues -split ';')) {
            $WarningText += "- $KnownIssue`r`n"
        }
        New-Popup -Title $WarningTitle -Message $WarningText -Buttons OK -Icon Exclamation | Out-Null
        Write-Log -Message "Known issues acknowledged" -Source "Init" -MessageType Info
    }

    # Add components to bring window to front
    try {[SFW] | Out-Null} catch {
        Add-Type @"
        using System;
        using System.Runtime.InteropServices;
        public class SFW {
            [DllImport("user32.dll")]
            [return: MarshalAs(UnmanagedType.Bool)]
            public static extern bool SetForegroundWindow(IntPtr hWnd);
        }
"@}

    $InstanceRunningInCurrentProcess = Get-Runspace | Where-Object {$_.Name -match "DataCollection_Interface"}
    $InstanceRunningInOtherProcess = Get-Process -Name Powershell -ErrorAction SilentlyContinue | Where-Object {$_.MainWindowTitle -eq "Microsoft Office 365 - Network Performance Assessment Tool"}
    if ($InstanceRunningInCurrentProcess) {
        Write-Error "Another instance of the tool is already running in the current Powershell session (PID:$PID)"
        Write-Log -Message "Another instance of the tool is already running in the current Powershell session (PID:$PID)" -Source "Init" -MessageType Info
        [SFW]::SetForegroundWindow((Get-Process -Id $PID).MainWindowHandle) | Out-Null
    } elseif ($InstanceRunningInOtherProcess) {
        Write-Error "Another instance of the tool is already running in another Powershell session (PID:$($InstanceRunningInOtherProcess.Id))"
        Write-Log -Message "Another instance of the tool is already running in another Powershell session (PID:$($InstanceRunningInOtherProcess.Id))" -Source "Init" -MessageType Info
        [SFW]::SetForegroundWindow($InstanceRunningInOtherProcess.MainWindowHandle) | Out-Null
    } else {
        Write-Host "Initializing interface"
        Write-Log -Message "Starting interface initialization" -Source "Init" -MessageType Info
        $CurrentSettings = Get-Content $env:NPAPath\Settings.json | ConvertFrom-Json
        $DebugMode = if ($CurrentSettings.DebugMode -match "True") {$true} else {$false}
        $WindowStyle = if ($DebugMode) { "Normal" } else { "Hidden" }
        $Command = "& `'$env:NPAPath\Scripts\LoadDialog.ps1`'"
        $InterfaceProcess = Start-Process powershell -ArgumentList $Command -WindowStyle $WindowStyle -PassThru
        Start-Sleep -Seconds 1
        [SFW]::SetForegroundWindow((Get-Process -Id $InterfaceProcess.Id).MainWindowHandle) | Out-Null
    }
}
